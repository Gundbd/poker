cmake_minimum_required(VERSION 3.20)
project(PokerCV LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

find_package(OpenCV REQUIRED COMPONENTS core imgproc highgui videoio dnn)

add_executable(PokerCV src/main_cv.cpp)
target_include_directories(PokerCV PRIVATE ${CMAKE_SOURCE_DIR}/include)
target_link_libraries(PokerCV PRIVATE
    opencv_core
    opencv_imgproc
    opencv_highgui
    opencv_videoio
    opencv_dnn
)

add_executable(GameDetection src/game_detection.cpp)
target_include_directories(GameDetection PRIVATE ${CMAKE_SOURCE_DIR}/include)
target_link_libraries(GameDetection PRIVATE    
    opencv_core
    opencv_imgproc
    opencv_highgui
    opencv_videoio
    opencv_dnn
)

add_executable(PokerConsole src/main.cpp)
target_include_directories(PokerConsole PRIVATE
    ${CMAKE_SOURCE_DIR}/include
    ${CMAKE_SOURCE_DIR}
)

if(EXISTS "${CMAKE_SOURCE_DIR}/models/best.onnx")
    add_custom_command(TARGET PokerCV POST_BUILD
        COMMAND ${CMAKE_COMMAND} -E make_directory "$<TARGET_FILE_DIR:PokerCV>/models"
        COMMAND ${CMAKE_COMMAND} -E copy_if_different
            "${CMAKE_SOURCE_DIR}/models/best.onnx"
            "$<TARGET_FILE_DIR:PokerCV>/models/best.onnx"
        COMMENT "Copy models/best.onnx"
    )
endif()
